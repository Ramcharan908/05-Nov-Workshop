Target: PostgreSQL 15+ 
-- Enable UUID Generation Extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. Base Entity: users
CREATE TABLE IF NOT EXISTS users (
    uuid UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email TEXT NOT NULL UNIQUE,
    role TEXT NOT NULL CHECK (role IN ('student', 'organizer', 'admin'))
);

-- 2. Entity: events
CREATE TABLE IF NOT EXISTS events (
    event_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES users(uuid) ON DELETE CASCADE,
    title TEXT NOT NULL,
    capacity INT4 NOT NULL CHECK (capacity > 0),
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 3. Junction Entity: registrations (User <-> Event M:N)
CREATE TABLE IF NOT EXISTS registrations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID NOT NULL REFERENCES users(uuid) ON DELETE CASCADE,
    event_id UUID NOT NULL REFERENCES events(event_id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    UNIQUE (student_id, event_id)
);

-- High Performance Foreign Key Indices
CREATE INDEX idx_events_student_id ON events(student_id);
CREATE INDEX idx_registrations_student_id ON registrations(student_id);
CREATE INDEX idx_registrations_event_id ON registrations(event_id);
